# One executable per layer; every TEST_CASE becomes its own ctest entry.
function(kstone_unit_test name)
  add_executable(${name} ${ARGN})
  kstone_target_defaults(${name})
  if(KSTONE_SANITIZE STREQUAL "thread" AND KSTONE_SETARCH)
    # Kernels >= 6.6 default to 32-bit mmap entropy, which TSan's shadow layout cannot map.
    set_target_properties(${name} PROPERTIES CROSSCOMPILING_EMULATOR "${KSTONE_SETARCH};${CMAKE_SYSTEM_PROCESSOR};-R")
  endif()
  target_link_libraries(${name} PRIVATE kstone_test_main kstone::kstone)
  target_compile_definitions(${name} PRIVATE KSTONE_FIXTURES="${CMAKE_CURRENT_SOURCE_DIR}/fixtures")
  doctest_discover_tests(${name} TEST_PREFIX "${name}/")
endfunction()
